```python
    def _log_system_warning(self, response: requests.Response) -> None:
        """Log KSeF X-System-Warning header (v2.6.0+) once per distinct value."""
        raw = response.headers.get("X-System-Warning")
        if not raw:
            return
        # Nagłówek pochodzi z zewnątrz — bez znaków sterujących (log injection), z limitem długości
        value = "".join(ch if ch.isprintable() else " " for ch in str(raw))
        value = value[: self.SYSTEM_WARNING_MAX_LEN]
        if value in self._seen_system_warnings:
            return
        self._seen_system_warnings.add(value)
        logger.warning("KSeF X-System-Warning: %s", value)
```

```python
"""Sonda granicy dateRange (KSeF TEST). Uruchom: python probe_date_range.py /ścieżka/config_test.json"""
import sys
from datetime import datetime, timedelta, timezone

sys.path.insert(0, ".")
from app.config_manager import ConfigManager
from app.ksef_client import KSeFClient

client = KSeFClient(ConfigManager(sys.argv[1]))
assert "api-test" in client.base_url, f"Not TEST: {client.base_url}"
assert client.authenticate(), "auth failed"

url = f"{client.base_url}/{client.API_VERSION}/invoices/query/metadata"
to = datetime.now(timezone.utc).replace(microsecond=0)
spans = [timedelta(days=89), timedelta(days=99), timedelta(days=100) - timedelta(seconds=1),
         timedelta(days=100), timedelta(days=100, seconds=1), timedelta(days=101)]
for span in spans:
    payload = {"subjectType": "Subject1", "dateRange": {
        "dateType": "Invoicing",
        "from": (to - span).strftime("%Y-%m-%dT%H:%M:%S.000Z"),
        "to": to.strftime("%Y-%m-%dT%H:%M:%S.000Z")}}
    r = client._make_authenticated_request("POST", url, params={"pageSize": 10, "pageOffset": 0},
                                           headers={"Content-Type": "application/json"}, json=payload)
    detail = "" if r is None or r.ok else client._extract_api_error_details(r)
    print(f"{span}: {None if r is None else r.status_code} {detail}")
```
